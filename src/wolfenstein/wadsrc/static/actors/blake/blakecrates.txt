// Blastable crates (codes 432-485). The code offset from the crate base
// selects the content, which spills onto the crate's tile when it is
// destroyed. In Planet Strike the crate simply breaks open; in Aliens of
// Gold it also explodes with radius damage, chaining adjacent crates.

actor BreakableCrate1 : SmartAnimation
{
	radius 32
	health 1
	+SOLID
	+SHOOTABLE
	states
	{
		Spawn:
			CRT1 A -1
			stop
		Death:
			EXPL B 0 bright A_AlertMonsters
			EXPL B 0 bright A_InitSmartAnim(random(3, 10))
			EXPL BCDE 1.5 bright A_SmartAnimDelay
			stop
	}
}

actor BlastableCrate1 : BreakableCrate1
{
	deathsound "crate/explode"
	states
	{
		Death:
			EXPL B 0 bright A_Scream
			EXPL B 0 bright A_Explode(20, 128, XF_HURTSOURCE, true)
			EXPL B 0 bright A_InitSmartAnim(random(24, 31))
			EXPL BCDE 1.5 bright A_SmartAnimDelay
			stop
	}
}

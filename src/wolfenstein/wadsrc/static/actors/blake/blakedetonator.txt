actor PlasmaDetonator : Ammo
{
	inventory.amount 1
	inventory.maxamount 100
	inventory.pickupsound "misc/plasmadetonator/pickup"
	states
	{
		Spawn:
			DBOM A -1
			stop
	}
}

actor PlasmaDetonatorWeapon : Weapon
{
	inventory.pickupsound "misc/plasmadetonator/pickup"
	weapon.ammotype1 "PlasmaDetonator"
	weapon.ammogive1 1
	weapon.ammouse1 0
	+WEAPON.NOAUTOFIRE

	action native A_BlakeDropDetonator();

	states
	{
		Ready:
			TNT1 A 1 A_WeaponReady
			loop
		Select:
			TNT1 A 0 A_Raise
			loop
		Deselect:
			TNT1 A 0 A_Lower
			loop
		Fire:
			TNT1 A 10 A_BlakeDropDetonator
			goto Ready
		Spawn:
			DBOM A -1
			stop
	}
}

actor PlasmaDetonatorDrop : SmartAnimation
{
	health 1
	+SHOOTABLE
	deathsound "misc/plasmadetonator/explosion"

	action native A_BlakeDetonatorExplode();

	states
	{
		Spawn:
			DBOM ABABABABABABABABABABABAB 10
			goto Death
		Death:
			PEXP A 0 A_Scream
			PEXP A 0 A_Explode(500, 128, XF_HURTSOURCE, true)
			PEXP A 0 A_BlakeDetonatorExplode
			PEXP A random(1.5, 3) A_InitSmartAnim(random(3, 6))
			PEXP BCDEFGH 1.5 A_SmartAnimDelay
			stop
	}
}

actor SecurityCube
{
	health 175
	+SOLID

	}

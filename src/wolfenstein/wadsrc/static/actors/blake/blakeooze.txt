actor GreenOoze : SmartAnimation
{
	states
	{
		Spawn:
			// Spawn-frame actions are skipped, so lead with a no-op frame.
			OOZ1 A 0
			OOZ1 A 0 A_InitSmartAnim(random(5, 7))
			OOZ1 A 0 A_Jump(256, 1, 2, 3)
		Loop:
			OOZ1 AB 2.5 A_SmartAnimDelay
			OOZ1 C 2.5 A_SmartAnimDelay A_OozeDamage
			loop
	}
}
